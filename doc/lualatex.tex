\section*{5 Formulas from physics and mathematics}

% P01: Analytical mechanics; regular Lagrangian and canonical variables.
\[
\begin{aligned}
\frac{\mathrm d}{\mathrm dt}\frac{\partial L}{\partial\dot q_i}
 -\frac{\partial L}{\partial q_i}&=0,
 & p_i&=\frac{\partial L}{\partial\dot q_i},\\[4pt]
\dot q_i&=\frac{\partial H}{\partial p_i},
 & \dot p_i&=-\frac{\partial H}{\partial q_i}.
\end{aligned}
\]

% P02: Maxwell equations, microscopic SI form (rho and j are total sources).
\[
\left\{
\begin{aligned}
\nabla\cdot\symbfit{E}&=\frac{\rho}{\varepsilon_0},\\[4pt]
\nabla\cdot\symbfit{B}&=0,\\[4pt]
\nabla\times\symbfit{E}&=-\frac{\partial\symbfit{B}}{\partial t},\\[4pt]
\nabla\times\symbfit{B}&=\mu_0\symbfit{j}
  +\mu_0\varepsilon_0\frac{\partial\symbfit{E}}{\partial t}.
\end{aligned}
\right.
\]

% P03: Nonrelativistic spinless particle in a scalar potential.
\[
i\hbar\frac{\partial\psi}{\partial t}
 =\hat H\psi
 =\left[-\frac{\hbar^2}{2m}\nabla^2+V(\symbfit{r},t)\right]\psi.
\]

% P04: Quantum canonical ensemble; assume the partition function exists.
\[
\hat\rho=\frac{e^{-\beta\hat H}}{Z},\qquad
Z=\operatorname{tr}\!\left(e^{-\beta\hat H}\right),\qquad
\beta=\frac{1}{k_{\mathrm B}T}.
\]

% P05: Einstein field equation with cosmological constant.
\[
R_{\mu\nu}-\frac12 Rg_{\mu\nu}+\Lambda g_{\mu\nu}
 =\frac{8\pi G}{c^4}T_{\mu\nu}.
\]

% P06: Incompressible Newtonian fluid with constant rho and dynamic viscosity mu.
% f is body force per unit mass.
\[
\begin{aligned}
\rho\left(\frac{\partial\symbfit{u}}{\partial t}
 +(\symbfit{u}\cdot\nabla)\symbfit{u}\right)
 &=-\nabla p+\mu\nabla^2\symbfit{u}+\rho\symbfit{f},\\
\nabla\cdot\symbfit{u}&=0.
\end{aligned}
\]

% M01: Retain r52 S5_11 verbatim, except its subsection title.
% A_n are events in one probability space; independence is not required.
\begin{equation*}
\begin{aligned}
\sum_{n=1}^{\infty}\symbb{P}(A_n)<\infty
\quad\Longrightarrow\quad
\symbb{P}\!\left(\limsup_{n\to\infty}A_n\right)
&=\symbb{P}\!\left(
 \bigcap_{m=1}^{\infty}\bigcup_{n=m}^{\infty}A_n
 \right)\\
&=0.
\end{aligned}
\end{equation*}

\sum_{k=0}^{n}(-1)^k\binom{n}{k}^{2}
=\begin{cases}

 0,&n\text{ odd}.
\end{cases}
\end{equation*}
